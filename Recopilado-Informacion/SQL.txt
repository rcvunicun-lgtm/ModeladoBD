Forma 1 de hacer un JOIN

SELECT OrderID, CustomerName, ShipperName   FROM (
SELECT Orders.OrderID, Customers.CustomerName, ShipperID 
FROM Orders
INNER JOIN Customers
	ON Orders.CustomerID = Customers.CustomerID
) AS tabla
INNER JOIN Shippers ON tabla.ShipperID = Shippers.ShipperID


Forma 2 de hacer EL MISMO JOIN

SELECT Orders.OrderID, Customers.CustomerName, Shippers.ShipperName
FROM ((Orders
INNER JOIN Customers ON Orders.CustomerID = Customers.CustomerID)
INNER JOIN Shippers ON Orders.ShipperID = Shippers.ShipperID);

SELECT MAX(EMPLEADO) AS [ID EMPLEADO], SUM([CANT VECES ATENDIDO]) AS [TOTAL DE VENTAS] FROM 
(
SELECT MAX(EmployeeID) AS [EMPLEADO], MAX(CustomerID) AS [CLIENTE], COUNT(CustomerID) AS [CANT VECES ATENDIDO] FROM Orders
GROUP BY EmployeeID, CustomerID
ORDER BY EmployeeID, CustomerID
)
GROUP BY EMPLEADO

SELECT MAX(EmployeeID) AS EMPLEADO, MAX(CustomerID) AS CLIENTE, COUNT(CustomerID) AS CANT FROM Orders

Cuatro formas distintas de obtener el mismo resultado en acces sql

SELECT Suppliers.SupplierID, Suppliers.SupplierName
FROM Suppliers
WHERE EXISTS (SELECT Products.SupplierID FROM Products WHERE Suppliers.SupplierID = Products.SupplierID AND Price < 20)


SELECT DISTINCT Products.SupplierID, Suppliers.SupplierName
FROM Products
INNER JOIN Suppliers ON Products.SupplierID = Suppliers.SupplierID
where Price < 20


SELECT Products.SupplierID, MAX(SupplierName), MIN(Price)
FROM Products
INNER JOIN Suppliers ON Products.SupplierID = Suppliers.SupplierID
GROUP BY Products.SupplierID
HAVING MIN(Price) < 20


SELECT Products.SupplierID, MAX(SupplierName)
FROM Products
INNER JOIN Suppliers ON Products.SupplierID = Suppliers.SupplierID
WHERE Price < 20
GROUP BY Products.SupplierID
